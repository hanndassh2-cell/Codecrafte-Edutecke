                            <input type="color" value={activeTemplate.sideText?.left?.color || '#000000'} onChange={(e) => setActiveTemplate({ ...activeTemplate, sideText: { ...activeTemplate.sideText, left: { ...activeTemplate.sideText?.left, color: e.target.value } as any } })} className="w-full h-9 rounded cursor-pointer" />
                          </div>
                          <div>
                            <label className="block text-slate-600 dark:text-slate-400 font-medium mb-1">الإزاحة الجانبية (عن حافة الصفحة)</label>
                            <input type="text" placeholder="مثال: 5px" value={activeTemplate.sideText?.left?.margin || ''} onChange={(e) => setActiveTemplate({ ...activeTemplate, sideText: { ...activeTemplate.sideText, left: { ...activeTemplate.sideText?.left, margin: e.target.value } as any } })} className="w-full p-2 rounded-lg border border-slate-200 dark:border-slate-700 dark:bg-slate-900 text-slate-900 dark:text-slate-100" />
                          </div>
                        </div>
                      </div>
                    </div>
                  )}
                </div>
              </div>
              </div>
            )}
          </div>

          {/* Lesson Content Paragraph Editor */}
          <div className="bg-white dark:bg-slate-900 p-5 rounded-2xl border border-slate-200 dark:border-slate-800 shadow-2xs space-y-4">
            <div className="flex items-center justify-between border-b pb-2">
              <h3 className="text-xs font-bold text-slate-900 dark:text-white flex items-center gap-2">
                <BookOpenCheck className="w-4 h-4 text-emerald-600" />
                <span>فقرات المحتوى الشرحي والنظري</span>
              </h3>
              <button
                onClick={addParagraph}
                className="px-2.5 py-1 rounded-lg bg-blue-50 text-blue-700 text-xs font-bold hover:bg-blue-100 transition flex items-center gap-1"
              >
                <Plus className="w-3.5 h-3.5" />
                <span>إضافة فقرة</span>
              </button>
            </div>

            <div className="space-y-3">
              {paragraphs.map((p, idx) => (
                <div
                  key={p.id}
                  draggable
                  onDragStart={(e) => handleDragStart(e, idx)}
                  onDragOver={(e) => handleDragOver(e, idx)}
                  onDragEnd={handleDragEnd}
                  className={`p-3 bg-slate-50 dark:bg-slate-800/60 rounded-xl border space-y-2 text-xs transition-shadow cursor-default ${
                    draggedIdx === idx
                      ? "border-blue-500 shadow-md opacity-50"
                      : "border-slate-200 dark:border-slate-700"
                  }`}
                >
                  <div className="flex items-center justify-between gap-2">
                    <div className="flex items-center gap-1.5 flex-1">
                      <div className="cursor-grab active:cursor-grabbing p-1">
                        <GripVertical className="w-4 h-4 text-slate-400 shrink-0" />
                      </div>
                      {editingTitleId === p.id ? (
                        <input
                          type="text"
                          value={p.title}
                          onChange={(e) => {
                            const next = [...paragraphs];
                            next[idx].title = e.target.value;
                            setParagraphs(next);
                          }}
                          onBlur={() => setEditingTitleId(null)}
                          onKeyDown={(e) => {
                            if (e.key === 'Enter') setEditingTitleId(null);
                          }}
                          autoFocus
                          className="font-bold bg-transparent border-b border-blue-500 pb-0.5 text-slate-900 dark:text-slate-100 outline-none w-full transition"
                        />
                      ) : (
                        <div className="font-bold text-slate-900 dark:text-slate-100 flex-1 flex items-center gap-2 group">
                          <span className={`${p.style?.color || ''}`} style={{ fontFamily: p.style?.fontFamily, fontSize: p.style?.fontSize, color: p.style?.color && !p.style.color.startsWith('text-') ? p.style.color : undefined }}>{p.title}</span>
                          <button
                            onClick={() => setEditingTitleId(p.id)}
                            className="opacity-0 group-hover:opacity-100 text-slate-400 hover:text-blue-600 transition-opacity p-1"
                            title="تعديل اسم الفقرة"
                          >
                            <Edit2 className="w-3.5 h-3.5" />
                          </button>
                          <button
                            onClick={() => setStylingParagraphId(p.id)}
                            className="opacity-0 group-hover:opacity-100 text-slate-400 hover:text-purple-600 transition-opacity p-1"
                            title="تنسيق عنوان الفقرة"
                          >
                            <Palette className="w-3.5 h-3.5" />
                          </button>
                        </div>
                      )}
                    </div>
                    <div className="flex items-center gap-1 shrink-0">
                      <button
                        onClick={() => {
                          const next = new Set(collapsedParagraphIds);
                          if (next.has(p.id)) next.delete(p.id);
                          else next.add(p.id);
                          setCollapsedParagraphIds(next);
                        }}
                        title={collapsedParagraphIds.has(p.id) ? "إظهار المحتوى" : "طي المحتوى"}
                        className="text-slate-400 hover:text-blue-600 p-1"
                      >
                        {collapsedParagraphIds.has(p.id) ? <ChevronDown className="w-4 h-4" /> : <ChevronUp className="w-4 h-4" />}
                      </button>
                      <button
                        onClick={() => {
                          if (idx > 0) {
                            const next = [...paragraphs];
                            [next[idx - 1], next[idx]] = [next[idx], next[idx - 1]];
                            setParagraphs(next);
                          }
                        }}
                        disabled={idx === 0}
                        title="تحريك لأعلى"
                        className="text-slate-400 hover:text-blue-600 disabled:opacity-30 disabled:hover:text-slate-400 p-1"
                      >
                        <ChevronUp className="w-4 h-4" />
                      </button>
                      <button
                        onClick={() => {
                          if (idx < paragraphs.length - 1) {
                            const next = [...paragraphs];
                            [next[idx], next[idx + 1]] = [next[idx + 1], next[idx]];
                            setParagraphs(next);
                          }
                        }}
                        disabled={idx === paragraphs.length - 1}
                        title="تحريك لأسفل"
                        className="text-slate-400 hover:text-blue-600 disabled:opacity-30 disabled:hover:text-slate-400 p-1"
                      >
                        <ChevronDown className="w-4 h-4" />
                      </button>
                      <button
                        onClick={() => removeParagraph(p.id)}
                        title="حذف الفقرة"
                        className="text-red-500 hover:text-red-700 p-1"
                      >
                        <Trash2 className="w-4 h-4" />
                      </button>
                    </div>
                  </div>

                  {!collapsedParagraphIds.has(p.id) && (
                    <QuestionTextBuilder
                      value={p.body}
                    onChange={(val) => {
                      const next = [...paragraphs];
                      next[idx].body = val;
                      setParagraphs(next);
                    }}
                    title="محرر أسطر الفقرة"
                    description="محتوى الفقرة"
                    placeholderPrefix="محتوى الفقرة"
                      collapsible={false}
                      inputStyle={{
                        fontFamily: p.style?.bodyFontFamily,
                        fontSize: p.style?.bodyFontSize,
                      }}
                      customPreview={
                        <div 
                          className="text-slate-800"
                          style={{
                            fontFamily: p.style?.bodyFontFamily,
                            fontSize: p.style?.bodyFontSize,
                          }}
                        >
                          {p.body.split('\n').map((line, i) => (
                            <div key={i} style={{ minHeight: "1.5em" }}>
                              <MathText text={line} />
                            </div>
                          ))}
                        </div>
                      }
                    />
                  )}
                </div>
              ))}
            </div>
          </div>

          {/* Attached Questions from Question Bank Selector */}
          <div className="bg-white dark:bg-slate-900 p-5 rounded-2xl border border-slate-200 dark:border-slate-800 shadow-2xs space-y-4">
            <div className="flex items-center justify-between border-b pb-2">
              <h3 className="text-xs font-bold text-slate-900 dark:text-white flex items-center gap-2">
                <Database className="w-4 h-4 text-purple-600" />
                <span>الأسئلة المدرجة في الدرس (من بنك الأسئلة)</span>
              </h3>
              <div className="flex items-center gap-2">
                <button
                  onClick={() => setIsAddingQuestion(!isAddingQuestion)}
                  className="px-2 py-1 text-[10px] font-bold rounded bg-blue-50 text-blue-600 hover:bg-blue-100 dark:bg-blue-900/30 dark:text-blue-400"
                >
                  {isAddingQuestion ? "إلغاء الإضافة" : "+ إضافة سؤال"}
                </button>
                <button
                  onClick={() => setIsQuestionsCollapsed(!isQuestionsCollapsed)}
                  className="text-slate-400 hover:text-blue-600 p-1"
                >
                  {isQuestionsCollapsed ? <ChevronDown className="w-4 h-4" /> : <ChevronUp className="w-4 h-4" />}
                </button>
              </div>
            </div>

            {!isQuestionsCollapsed && (
              <div className="space-y-4">
                {/* Selected Questions List (Draggable) */}
                {attachedQuestions.length > 0 ? (
                  <div className="space-y-2">
                    {attachedQuestions.map((q, idx) => (
                      <div
                        key={q.id}
                        draggable
                        onDragStart={(e) => {
                          setDraggedQuestionIdx(idx);
                          e.dataTransfer.effectAllowed = "move";
                          e.dataTransfer.setData("text/html", e.currentTarget.parentNode as any);
                        }}
                        onDragOver={(e) => {
                          e.preventDefault();
                          if (draggedQuestionIdx === null || draggedQuestionIdx === idx) return;
                          const next = [...selectedQuestionIds];
                          const draggedItem = next[draggedQuestionIdx];
                          next.splice(draggedQuestionIdx, 1);
                          next.splice(idx, 0, draggedItem);
                          setDraggedQuestionIdx(idx);
                          setSelectedQuestionIds(next);
                        }}
                        onDragEnd={() => setDraggedQuestionIdx(null)}
                        className={`p-3 bg-slate-50 dark:bg-slate-800/60 rounded-xl border flex items-center justify-between gap-3 transition-shadow ${
                          draggedQuestionIdx === idx
                            ? "border-blue-500 shadow-md opacity-50"
                            : "border-slate-200 dark:border-slate-700"
                        }`}
                      >
                        <div className="flex items-center gap-2 flex-1 min-w-0">
                          <div className="cursor-grab active:cursor-grabbing p-1 shrink-0">
                            <GripVertical className="w-4 h-4 text-slate-400" />
                          </div>
                          <div className="flex-1 min-w-0">
                            <div className="flex items-center gap-2 mb-1">
                              <span className="px-2 py-0.5 rounded text-[10px] font-bold bg-blue-100 text-blue-800 dark:bg-blue-900/50 dark:text-blue-300">
                                {QUESTION_TYPES_LABEL[q.type] || q.type.toUpperCase()}
                              </span>
                            </div>
                            <div className="font-semibold text-slate-900 dark:text-slate-100 text-xs truncate">
                              {q.text}
                            </div>
                          </div>
                        </div>
                        <div className="flex items-center gap-1 shrink-0">
                          <button
                            onClick={() => {
                              if (idx > 0) {
                                const next = [...selectedQuestionIds];
                                [next[idx - 1], next[idx]] = [next[idx], next[idx - 1]];
                                setSelectedQuestionIds(next);
                              }
                            }}
                            disabled={idx === 0}
                            title="تحريك لأعلى"
                            className="text-slate-400 hover:text-blue-600 disabled:opacity-30 p-1"
                          >
                            <ChevronUp className="w-4 h-4" />
                          </button>
                          <button
                            onClick={() => {
                              if (idx < selectedQuestionIds.length - 1) {
                                const next = [...selectedQuestionIds];
                                [next[idx], next[idx + 1]] = [next[idx + 1], next[idx]];
                                setSelectedQuestionIds(next);
                              }
                            }}
                            disabled={idx === selectedQuestionIds.length - 1}
                            title="تحريك لأسفل"
                            className="text-slate-400 hover:text-blue-600 disabled:opacity-30 p-1"
                          >
                            <ChevronDown className="w-4 h-4" />
                          </button>
                          <button
                            onClick={() => setSelectedQuestionIds(selectedQuestionIds.filter(id => id !== q.id))}
                            title="إزالة السؤال"
                            className="text-red-500 hover:text-red-700 p-1"
                          >
                            <Trash2 className="w-4 h-4" />
                          </button>
                        </div>
                      </div>
                    ))}
                  </div>
                ) : (
                  <div className="text-center py-6 text-xs text-slate-500 border border-dashed rounded-xl border-slate-300 dark:border-slate-700">
                    لا يوجد أسئلة مدرجة. انقر على "إضافة سؤال" لاختيار أسئلة من بنك الأسئلة.
                  </div>
                )}

                {/* Available Questions List (Add Mode) */}
                {isAddingQuestion && (
                  <div className="mt-4 pt-4 border-t border-slate-200 dark:border-slate-800">
                    <h4 className="text-[11px] font-bold text-slate-600 mb-3">اختر أسئلة للإضافة:</h4>
                    <div className="space-y-2 text-xs max-h-[36rem] overflow-y-auto pr-2">
                      {availableQuestions.map((q) => {
                        const isChecked = selectedQuestionIds.includes(q.id);
                        return (
                          <label
                            key={q.id}
                            className={`flex items-start gap-2.5 p-2.5 rounded-xl border transition cursor-pointer ${
                              isChecked
                                ? "bg-blue-50 dark:bg-blue-900/30 border-blue-300 dark:border-blue-700 opacity-50"
                                : "bg-slate-50 dark:bg-slate-800 border-slate-200 dark:border-slate-700 hover:border-blue-300"
                            }`}
                          >
                            <input
                              type="checkbox"
                              checked={isChecked}
                              onChange={(e) => {
                                if (e.target.checked) {
                                  setSelectedQuestionIds([...selectedQuestionIds, q.id]);
                                } else {
                                  setSelectedQuestionIds(
                                    selectedQuestionIds.filter((id) => id !== q.id)
                                  );
                                }
                              }}
                              className="mt-0.5 rounded text-blue-600 focus:ring-blue-500"
                            />
                            <div>
                              <div className="flex items-center gap-2 mb-1">
                                <span className="px-2 py-0.5 rounded text-[10px] font-bold bg-blue-100 text-blue-800 dark:bg-blue-900/50 dark:text-blue-300">
                                  {QUESTION_TYPES_LABEL[q.type] || q.type.toUpperCase()}
                                </span>
                              </div>
                              <div className="font-semibold text-slate-900 dark:text-slate-100">
                                {q.text}
                              </div>
                              <div className="text-[10px] text-slate-500 mt-0.5">
                                الأهمية: {q.importance}/5 • الصعوبة: {q.difficulty} • احتمال الورود: {q.futureProbability}%
                              </div>
                            </div>
                          </label>
                        );
                      })}
                    </div>
                  </div>
                )}
              </div>
            )}
          </div>
        </div>
        {/* Right Side: Live Printable A4 Preview Sheet (6 cols) */}
        <div className={`lg:col-span-6 sticky top-20 transition-all ${isFullscreen ? 'fixed inset-0 z-50 bg-slate-900/90 flex flex-col items-center justify-start overflow-y-auto p-4 sm:p-8 backdrop-blur-sm' : ''}`}>
          <div className="flex items-center justify-between mb-2 w-full max-w-[210mm] mx-auto">
            <div className="text-xs font-bold text-slate-500 flex items-center gap-2">
              <Printer className="w-4 h-4 text-blue-600" />
              <span className={isFullscreen ? "text-slate-200" : ""}>معاينة نموذج الطباعة الرسمي الفعلي</span>
            </div>
            
            <div className="flex items-center gap-1.5">
              <button
                onClick={() => setZoomLevel(Math.max(0.5, zoomLevel - 0.1))}
                className={`p-1.5 rounded-lg ${isFullscreen ? 'text-slate-300 hover:bg-slate-800' : 'text-slate-500 hover:bg-slate-200 dark:hover:bg-slate-800'} transition`}
                title="تصغير"
              >
                <ZoomOut className="w-4 h-4" />
              </button>
              <span className={`text-[10px] font-bold ${isFullscreen ? 'text-slate-300' : 'text-slate-500'} w-8 text-center`}>
                {Math.round(zoomLevel * 100)}%
              </span>
              <button
                onClick={() => setZoomLevel(Math.min(2, zoomLevel + 0.1))}
                className={`p-1.5 rounded-lg ${isFullscreen ? 'text-slate-300 hover:bg-slate-800' : 'text-slate-500 hover:bg-slate-200 dark:hover:bg-slate-800'} transition`}
                title="تكبير"
              >
                <ZoomIn className="w-4 h-4" />
              </button>
              <div className="w-px h-4 bg-slate-300 dark:bg-slate-700 mx-1"></div>
              <button
                onClick={() => setIsFullscreen(!isFullscreen)}
                className={`p-1.5 rounded-lg ${isFullscreen ? 'bg-blue-600 text-white hover:bg-blue-700' : 'text-slate-500 hover:bg-slate-200 dark:hover:bg-slate-800'} transition`}
                title={isFullscreen ? "تصغير الشاشة" : "ملء الشاشة"}
              >
                <Maximize2 className="w-4 h-4" />
              </button>
            </div>
          </div>

          <div className="w-full overflow-x-auto pb-8">
            <div 
              className="origin-top flex justify-center min-w-[794px] mx-auto"
              style={{ transform: `scale(${zoomLevel})`, transformOrigin: 'top center', marginBottom: `${(zoomLevel - 1) * 100}%` }}
            >
              <PaginatedA4Preview
            template={{
              ...activeTemplate,
              headerContent: {
                ...activeTemplate.headerContent,
                subjectName: currentSubject?.name || "اسم المادة",
              },
              footerContent: {
                ...activeTemplate.footerContent
              }
            }}
            title={currentLesson?.title}
            subtitle={`الأهداف التعليمية: ${currentLesson?.objectives?.join(" • ") || "مخرجات المنهاج المعتمدة"}`}
            hierarchyText={`${currentUnit?.title || "الوحدة"}\n${currentLesson?.title || "الدرس"} | ${currentLesson?.durationMinutes ? currentLesson.durationMinutes + " دقيقة" : "45 دقيقة"}`}
            items={[
              ...paragraphs.map((p) => ({
                id: p.id,
                content: (
                  <section className="space-y-1 mb-4">
                    <h3 
                      className={`text-xs font-extrabold border-b border-blue-200 pb-0.5 ${p.style?.color || 'text-blue-900'}`}
                      style={{
                        fontFamily: p.style?.fontFamily,
                        fontSize: p.style?.fontSize,
                        color: p.style?.color && !p.style.color.startsWith('text-') ? p.style.color : undefined
                      }}
                    >
                      {p.title}
                    </h3>
                    <div className="lesson-content-paragraph text-[11px] leading-relaxed text-slate-800" style={{ fontFamily: p.style?.bodyFontFamily, fontSize: p.style?.bodyFontSize }}>
                      {p.body.split('\n').map((line, i) => (
                        <div key={i} style={{ minHeight: "1.5em" }}>
                          <MathText text={line} />
                        </div>
                      ))}
                    </div>
                  </section>
                )
              })),
              ...(attachedQuestions.length > 0 ? [{
                id: 'questions-section-header',
                content: (
                  <h3 className="text-xs font-bold text-slate-900 bg-slate-100 p-1.5 rounded mb-3 mt-4 border-t-2 border-slate-800 pt-3">
                    أسئلة وتطبيقات الدرس المقترحة:
                  </h3>
                )
              }] : []),
              ...attachedQuestions.map((q, idx) => ({
                id: q.id,
                content: (
                  <div className="text-[11px] p-2.5 rounded border border-slate-200 bg-slate-50/50 space-y-1.5 mb-3">
                    <div className="font-bold text-slate-900 flex items-baseline gap-1">
                      <span className="shrink-0">س {idx + 1}:</span>
                      <MathText text={q.text} inline />
                    </div>
                    {q.type === "mcq" && q.distractors && (
                      <div className="grid grid-cols-2 gap-1.5 pr-3 text-[10px] text-slate-700">
                        {q.distractors.map((d, dIdx) => (
                          <div key={d.id} className="flex items-baseline gap-1">
                            <span className="shrink-0 font-bold">{["أ", "ب", "ج", "د"][dIdx] || "-"}:</span>
                            <MathText text={d.text} inline />
                          </div>
                        ))}
                      </div>
                    )}
                    {q.type === "true_false" && (
                      <div className="pr-3 text-[10px] text-slate-700 font-semibold">
                        [ ] صح &nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp; [ ] خطأ
                      </div>
                    )}
                    <div className="text-[10px] text-slate-500 pt-1 border-t border-slate-200 flex items-baseline gap-1">
                      <span className="shrink-0 font-bold">الإجابة النموذجية:</span>
                      <MathText text={q.answer} inline />
                    </div>
                  </div>
                )
              }))
            ]}
          />
            </div>
          </div>
        </div>
      </div>
    </div>
