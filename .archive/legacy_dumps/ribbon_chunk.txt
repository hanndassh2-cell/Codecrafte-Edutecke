                  className="px-3 py-1.5 bg-purple-600 hover:bg-purple-700 text-white rounded-lg font-bold text-xs flex items-center gap-1.5 cursor-pointer shadow-2xs transition-all"
                  title="المساعد الذكي لتحليل المحتوى والتدقيق"
                >
                  <Sparkles className="w-3.5 h-3.5" />
                  <span>المساعد الذكي والتدقيق</span>
                </button>

                <button
                  onMouseDown={(e) => e.preventDefault()}
                  onClick={() => setIsQuickPasteModalOpen(true)}
                  className="px-3 py-1.5 bg-white dark:bg-slate-900 hover:bg-purple-50 dark:hover:bg-purple-900/30 text-purple-700 dark:text-purple-300 border border-slate-300 dark:border-slate-600 rounded-lg font-bold text-xs flex items-center gap-1.5 cursor-pointer shadow-2xs"
                  title="تحليل اللصق الذكي"
                >
                  <Wand2 className="w-4 h-4 text-purple-600" />
                  <span>محلل النص الذكي</span>
                </button>

                <button
                  onMouseDown={(e) => e.preventDefault()}
                  onClick={() => setIsQuestionParserOpen(true)}
                  className="px-3 py-1.5 bg-white dark:bg-slate-900 hover:bg-purple-50 dark:hover:bg-purple-900/30 text-purple-700 dark:text-purple-300 border border-slate-300 dark:border-slate-600 rounded-lg font-bold text-xs flex items-center gap-1.5 cursor-pointer shadow-2xs"
                  title="استخراج وحذف الأسئلة من الدرس"
                >
                  <CheckSquare className="w-4 h-4 text-purple-600" />
                  <span>مستخرج الأسئلة</span>
                </button>
              </div>
              <span className="text-[10px] font-bold text-slate-500 dark:text-slate-400 text-center w-full uppercase tracking-wider">التدقيق والتحليل الذكي</span>
            </div>

            {/* Word & Content Stats Group */}
            <div className="flex flex-col justify-between bg-slate-50 dark:bg-slate-800/60 p-2 rounded-xl border border-slate-200/80 dark:border-slate-700/80 shadow-2xs shrink-0">
              <div className="flex items-center gap-3 px-2 py-1 mb-1">
                <div className="flex flex-col text-center">
                  <span className="text-[10px] font-bold text-slate-400">إجمالي البطاقات</span>
                  <span className="text-sm font-black text-blue-600">{paragraphs.length}</span>
                </div>
                <div className="w-px h-6 bg-slate-200 dark:bg-slate-700"></div>
                <div className="flex flex-col text-center">
                  <span className="text-[10px] font-bold text-slate-400">الأسئلة المرفقة</span>
                  <span className="text-sm font-black text-emerald-600">{attachedQuestions?.length || 0}</span>
                </div>
              </div>
              <span className="text-[10px] font-bold text-slate-500 dark:text-slate-400 text-center w-full uppercase tracking-wider">إحصائيات الدرس</span>
            </div>
          </>
        )}

        {/* TAB 7 CONTENT: VIEW (عرض) */}
        {activeRibbonTab === "view" && (
          <>
            {/* Display Modes & Fullscreen Group */}
            <div className="flex flex-col justify-between bg-slate-50 dark:bg-slate-800/60 p-2 rounded-xl border border-slate-200/80 dark:border-slate-700/80 shadow-2xs shrink-0">
              <div className="flex items-center gap-1.5 mb-1">
                {onPreview && (
                  <button
                    onMouseDown={(e) => e.preventDefault()}
                    onClick={onPreview}
                    className="px-3 py-1.5 bg-white dark:bg-slate-900 hover:bg-indigo-50 text-indigo-700 dark:text-indigo-300 border border-slate-300 dark:border-slate-600 rounded-lg font-bold text-xs flex items-center gap-1.5 cursor-pointer shadow-2xs"
                    title="معاينة ملء الشاشة"
                  >

                    <span>معاينة الشاشة</span>
                  </button>
                )}

                {setIsFocusReadingMode && (
                  <button
                    onMouseDown={(e) => e.preventDefault()}
                    onClick={() => setIsFocusReadingMode(!isFocusReadingMode)}
                    className={`px-3 py-1.5 rounded-lg font-bold text-xs flex items-center gap-1.5 cursor-pointer transition-all border shadow-2xs ${
                      isFocusReadingMode
                        ? "bg-amber-600 text-white border-amber-700"
                        : "bg-white dark:bg-slate-900 text-slate-700 dark:text-slate-200 border-slate-300 dark:border-slate-600 hover:bg-slate-100"

                    title="وضع القراءة الهادئ بدون تشتيت"

                  </button>
                )}

                {onToggleNav && (
                  <button
                    onMouseDown={(e) => e.preventDefault()}
                    onClick={onToggleNav}
                    className={`px-3 py-1.5 rounded-lg font-bold text-xs flex items-center gap-1.5 cursor-pointer transition-all border shadow-2xs ${
                      isNavCollapsed
                        ? "bg-white dark:bg-slate-900 text-slate-700 border-slate-300"
                        : "bg-blue-600 text-white border-blue-700"

                    title="إظهار/إخفاء القائمة الجانبية للمنهج"
                  >
                    <PanelLeft className="w-4 h-4" />
                    <span>شريط المنهج</span>
                  </button>
                )}

                {onTogglePreview && (
                  <button
                    onMouseDown={(e) => e.preventDefault()}
                    onClick={onTogglePreview}
                    className={`px-3 py-1.5 rounded-lg font-bold text-xs flex items-center gap-1.5 cursor-pointer transition-all border shadow-2xs ${
                      isPreviewCollapsed
                        ? "bg-white dark:bg-slate-900 text-slate-700 border-slate-300"
                        : "bg-emerald-600 text-white border-emerald-700"
                    }`}
                    title="إظهار/إخفاء شريط المعاينة المباشرة"
                  >
                    <PanelRight className="w-4 h-4" />
                    <span>لوحة المعاينة</span>
                  </button>
                )}
              </div>
              <span className="text-[10px] font-bold text-slate-500 dark:text-slate-400 text-center w-full uppercase tracking-wider">شاشات وألواح العرض</span>
            </div>

            {/* Ruler Controls Group */}
            <div className="flex flex-col justify-between bg-slate-50 dark:bg-slate-800/60 p-2 rounded-xl border border-slate-200/80 dark:border-slate-700/80 shadow-2xs shrink-0">
              <div className="flex items-center gap-1.5 mb-1">
                <button
                  onMouseDown={(e) => e.preventDefault()}
                  onClick={() => setShowRuler(!showRuler)}
                  className={`px-3 py-1.5 rounded-lg font-bold text-xs flex items-center gap-1.5 cursor-pointer transition-all border shadow-2xs ${
                    showRuler
                      ? "bg-purple-600 text-white border-purple-700"
                      : "bg-white dark:bg-slate-900 text-slate-700 dark:text-slate-200 border-slate-300 dark:border-slate-600 hover:bg-slate-100"

                  title="إظهار أو إخفاء المسطرة في أعلى المستند"
                >
                  <Ruler className="w-4 h-4" />
                  <span>المسطرة ({showRuler ? "ظاهرة" : "مخفية"})</span>
                </button>

                <button
                  onMouseDown={(e) => e.preventDefault()}
                  onClick={() => setIsRulerModalOpen(true)}
                  className="px-2.5 py-1.5 bg-white dark:bg-slate-900 hover:bg-purple-50 dark:hover:bg-slate-800 text-slate-700 dark:text-slate-200 border border-slate-300 dark:border-slate-600 rounded-lg font-bold text-xs flex items-center gap-1 cursor-pointer shadow-2xs"
                  title="فتح نافذة للتحكم بخواص المسطرة والهوامش"
                >
                  <Sliders className="w-4 h-4 text-purple-600" />
                  <span>خواص المسطرة</span>
                </button>
              </div>
              <span className="text-[10px] font-bold text-slate-500 dark:text-slate-400 text-center w-full uppercase tracking-wider">أدوات القياس والمسطرة</span>
            </div>

            {/* Zoom Controls Group */}
            {setZoomLevel && (
              <div className="flex flex-col justify-between bg-slate-50 dark:bg-slate-800/60 p-2 rounded-xl border border-slate-200/80 dark:border-slate-700/80 shadow-2xs shrink-0">
                <div className="flex items-center gap-1 mb-1">
                  <button
                    onMouseDown={(e) => e.preventDefault()}
                    onClick={() => setZoomLevel((z) => Math.max(50, z - 10))}

                  >
                    <ZoomOut className="w-4 h-4" />
                  </button>
                  <span className="px-2.5 py-1 text-xs font-black text-slate-800 dark:text-slate-100 bg-white dark:bg-slate-900 border border-slate-200 dark:border-slate-700 rounded-lg">
                    {zoomLevel}%
                  </span>
                  <button
                    onMouseDown={(e) => e.preventDefault()}
                    onClick={() => setZoomLevel((z) => Math.min(200, z + 10))}
                    className="p-2 rounded-lg bg-white dark:bg-slate-900 hover:bg-slate-100 text-slate-700 dark:text-slate-200 border border-slate-300 dark:border-slate-600 cursor-pointer"
                    title="تكبير (+10%)"
                  >
                    <ZoomIn className="w-4 h-4" />
                  </button>
                  <button
                    onMouseDown={(e) => e.preventDefault()}
                    onClick={() => setZoomLevel(100)}
                    className="px-2 py-1 text-[11px] font-bold rounded bg-slate-200 dark:bg-slate-800 hover:bg-slate-300 text-slate-700 dark:text-slate-200 cursor-pointer"
                    title="إعادة التكبير 100%"
                  >
                    100%
                  </button>
                </div>
                <span className="text-[10px] font-bold text-slate-500 dark:text-slate-400 text-center w-full uppercase tracking-wider">مستوى التكبير والتصغير</span>
              </div>
            )}
          </>
        )}

      </div>
      </div>
      )}
